import random
import time

class GameBola:
    def __init__(self):
        self.skor_kita = 0
        self.skor_lawan = 0
        self.menit = 0
        self.posisi_bola = "Tengah Lapangan"
        self.game_over = False

    def cetak_papan_skor(self):
        print("\n" + "="*45)
        print(f" 🏆 FINAL PIALA ASIA | ⏱ Menit: {self.menit}'")
        print(f" 🇮🇩 INDONESIA [{self.skor_kita}] VS [ {self.skor_lawan} ] JEPANG 🇯🇵")
        print(f" ⚽ Posisi Bola: {self.posisi_bola}")
        print("="*45)

    def serangan_lawan(self):
        print("\n⚠️ Jepang mulai menyusun serangan balik cepat!")
        time.sleep(1.5)
        pilihan_bertahan = input("Lawan menusuk! Pilih aksi bertahan:\n1. Tekel Bersih (Slide)\n2. Bayangi saja (Jockey)\nPilihanmu (1/2): ").strip()
        
        sukses = random.choice([True, False])
        if sukses:
            print("\n🔥 Mantap! Kamu berhasil merebut gila bola dan memotong operan mereka!")
            self.posisi_bola = "Tengah Lapangan"
        else:
            print("\n❌ Gawat! Pemain lawan berhasil melewati pertahanan kita...")
            time.sleep(1)
            if random.choice([True, False]):
                print("🥅 GOOOL untuk Jepang! Tendangan keras mereka tidak terbendung.")
                self.skor_lawan += 1
            else:
                print("🧤 Penyelamatan gemilang dari kiper kita! Bola aman.")
            self.posisi_bola = "Tengah Lapangan"

    def mainkan(self):
        print("==================================================")
        print("    WELCOME TO INDONESIA FOOTBALL CHAMPIONSHIP    ")
        print("==================================================")
        print("Bawa Timnas Garuda mencetak gol kemenangan!")
        time.sleep(1)

        while self.menit < 90 and not self.game_over:
            self.cetak_papan_skor()
            
            if self.posisi_bola == "Tengah Lapangan":
                print("Apa tindakanmu di tengah lapangan?")
                print("1. Oper pendek ke gelandang (Gaya Tiki-Taka)")
                print("2. Dribble solo melewati bek lawan")
                pilihan = input("Masukkan nomor (1/2): ").strip()
                
                if pilihan == "1":
                    print("\nOperan akurat! Bola sekarang mendekati kotak penalti lawan.")
                    self.posisi_bola = "Kotak Penalti Lawan"
                elif pilihan == "2":
                    if random.choice([True, False]):
                        print("\nSkill berkelas! Kamu berhasil melewati satu bek lawan dengan gaya!")
                        self.posisi_bola = "Kotak Penalti Lawan"
                    else:
                        print("\nBola berhasil direbut oleh gempuran pertahanan musuh.")
                        self.serangan_lawan()
                else:
                    print("\nKehilangan fokus! Bola direbut musuh.")
                    self.serangan_lawan()

            elif self.posisi_bola == "Kotak Penalti Lawan":
                print("Kamu sudah berada di depan gawang! Peluang emas!")
                print("1. Tendang langsung ke pojok gawang (Shooting)")
                print("2. Umpan silang ke striker lain (Crossing)")
                pilihan = input("Masukkan nomor (1/2): ").strip()
                
                if pilihan == "1":
                    print("\nKamu mengambil ancang-ancang... TENDANGG!!")
                    time.sleep(1.5)
                    if random.choice([True, True, False]): # Peluang gol lebih besar
                        print("🎉 GOOOOLLLLLL!! Tendangan melengkungmu merobek jala gawang musuh!")
                        self.skor_kita += 1
                    else:
                        print("💥 Sayang sekali! Tendanganmu membentur tiang gawang.")
                    self.posisi_bola = "Tengah Lapangan"
                elif pilihan == "2":
                    print("\nKamu melepaskan umpan lambung manis...")
                    time.sleep(1)
                    if random.choice([True, False]):
                        print("🎉 GOOOOLLLLLL!! Sundalan rekan setimmu menjebol gawang!")
                        self.skor_kita += 1
                    else:
                        print("Kiper musuh berhasil memotong umpan lambungmu.")
                    self.posisi_bola = "Tengah Lapangan"
                else:
                    print("\nTerdiam terlalu lama, bek musuh langsung menyapu bola.")
                    self.posisi_bola = "Tengah Lapangan"
            
            self.menit += random.randint(15, 25)
            time.sleep(1.5)

        # Akhir Pertandingan
        print("\n" + "=*="*15)
        print("          PRIIIIITTT! PERTANDINGAN SELESAI!          ")
        print(f"            SKOR AKHIR: INDONESIA {self.skor_kita} - {self.skor_lawan} JEPANG")
        print("=*="*15)
        
        if self.skor_kita > self.skor_lawan:
            print("🏆 LUAR BIASA! Kamu membawa INDONESIA JUARA ASIA! Angkat pialanya! 🇮🇩🥇")
        elif self.skor_kita == self.skor_lawan:
            print("🤝 Skor Imbang! Pertandingan berlanjut ke adu penalti di game berikutnya.")
        else:
            print("😢 Kamu kalah. Ayo latihan lagi dan balas dendam di turnamen berikutnya!")

if __name__ == "__main__":
    game = GameBola()
    game.mainkan()
